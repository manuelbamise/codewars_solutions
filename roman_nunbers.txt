Roman values work by comparing the current value to the next one.., if the current value is lower than the next value it is subtracted from the sumtotal.., if not, it is added.. By virtue of this logic, the last value is always added.


//CODE
func Decode(roman string) int {
	figureMap := map[rune]int{
		'I': 1,
		'V': 5,
		'X': 10,
		'L': 50,
		'C': 100,
		'D': 500,
		'M': 1000,
	}
	fmt.Println(figureMap)
	var output int

	for i := 0; i < len(roman); i++ {
		if i+1 < len(roman) && figureMap[rune(roman[i])] < figureMap[rune(roman[i+1])] {
			output -= figureMap[rune(roman[i])]
		} else {
			output += figureMap[rune(roman[i])]
		}
	}
