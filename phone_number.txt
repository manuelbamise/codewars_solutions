The problem this time around was to create a phone number out of an array of uint, format it by adding some brackets space and a dash, and return it as a string



// Code
func createPhoneNumber(numbers [10]uint) string {
	var newOutputString strings.Builder

	newOutputString.WriteString("(")
	for i, value := range numbers {
		newOutputString.WriteString(strconv.FormatUint(uint64(value), 10))

		if i == 2 {
			newOutputString.WriteString(") ")
		}

		if i == 5 {
			newOutputString.WriteString("-")
		}
	}
	return newOutputString.String()
}
